* RDM 334163 - Lê a MCHB uma vez e soma por material/lote
    SELECT matnr, werks, lgort, charg, clabs
      FROM mchb
      FOR ALL ENTRIES IN @it_mchbcar
     WHERE matnr EQ @it_mchbcar-matnr
       AND charg EQ @it_mchbcar-charg
       AND clabs GT 0
      INTO TABLE @DATA(lt_mchb_aux).
    LOOP AT lt_mchb_aux ASSIGNING FIELD-SYMBOL(<ls_mchb_aux>).
      COLLECT VALUE ty_c_mchb( matnr = <ls_mchb_aux>-matnr
                               charg = <ls_mchb_aux>-charg
                               clabs = <ls_mchb_aux>-clabs ) INTO lt_c_mchb.
    ENDLOOP.
    FREE lt_mchb_aux.
* RDM 334163 - Fim
