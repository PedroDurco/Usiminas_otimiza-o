*&---------------------------------------------------------------------*
*&      Form  ZF_GRAVA_SAIDA_LOTE
*&---------------------------------------------------------------------*
* RDM 334163 - Gravação da ZPPT_PAJT em uma única operação
*----------------------------------------------------------------------*
FORM zf_grava_saida_lote.

  CONSTANTS lc_cprog TYPE zppt_pajt-cprog VALUE 'ZSUPPR_PESQUISA_APLICACAO_LOTE'.

  DATA: lt_pajt   TYPE STANDARD TABLE OF zppt_pajt,
        lwa_pajt  TYPE zppt_pajt,
        lv_seq    TYPE zppt_pajt-seq,
        lv_varkey TYPE rstable-varkey.

  CHECK gt_outtab1[] IS NOT INITIAL OR gt_outtab2[] IS NOT INITIAL.

* Bloqueia a numeração do usuário enquanto grava (evita SEQ repetido entre jobs)
  lv_varkey(3)     = sy-mandt.
  lv_varkey+3(40)  = lc_cprog.
  lv_varkey+43(12) = sy-uname.

  CALL FUNCTION 'ENQUEUE_E_TABLE'
    EXPORTING
      mode_rstable   = 'E'
      tabname        = 'ZPPT_PAJT'
      varkey         = lv_varkey
      _wait          = abap_true
    EXCEPTIONS
      foreign_lock   = 1
      system_failure = 2
      OTHERS         = 3.
  DATA(lv_lock) = xsdbool( sy-subrc = 0 ).

  SELECT MAX( seq )
    FROM zppt_pajt
    INTO lv_seq
    WHERE cprog EQ lc_cprog
      AND ernam EQ sy-uname.

  LOOP AT gt_outtab1 INTO wa_outtab.
    ADD 1 TO lv_seq.
    CLEAR lwa_pajt.
    MOVE-CORRESPONDING wa_outtab TO lwa_pajt.
    lwa_pajt-cprog = lc_cprog.
    lwa_pajt-ernam = sy-uname.
    lwa_pajt-seq   = lv_seq.
    lwa_pajt-erdat = sy-datum.
    lwa_pajt-erzet = sy-uzeit.
    APPEND lwa_pajt TO lt_pajt.
  ENDLOOP.

  LOOP AT gt_outtab2 INTO wa_outtab.
    ADD 1 TO lv_seq.
    CLEAR lwa_pajt.
    MOVE-CORRESPONDING wa_outtab TO lwa_pajt.
    lwa_pajt-cprog = lc_cprog.
    lwa_pajt-ernam = sy-uname.
    lwa_pajt-seq   = lv_seq.
    lwa_pajt-erdat = sy-datum.
    lwa_pajt-erzet = sy-uzeit.
    APPEND lwa_pajt TO lt_pajt.
  ENDLOOP.

  INSERT zppt_pajt FROM TABLE lt_pajt ACCEPTING DUPLICATE KEYS.
  COMMIT WORK AND WAIT.

  IF lv_lock = abap_true.
    CALL FUNCTION 'DEQUEUE_E_TABLE'
      EXPORTING
        mode_rstable = 'E'
        tabname      = 'ZPPT_PAJT'
        varkey       = lv_varkey.
  ENDIF.

  FREE lt_pajt.

ENDFORM.                    " ZF_GRAVA_SAIDA_LOTE
* RDM 334163 - Fim
