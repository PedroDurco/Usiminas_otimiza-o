11B – Novo FORM

Onde: final do F01.

*&---------------------------------------------------------------------*
*&      Form  F_DESCARTA_COMBINACAO
*&---------------------------------------------------------------------*
* RDM 334163 - Ao primeiro vermelho: mantém só a mensagem vermelha da
*              combinação lote x SKU e remove as demais (amarelas anteriores)
*----------------------------------------------------------------------*
FORM f_descarta_combinacao USING p_ini_comb  TYPE i
                                 p_ini_chk   TYPE i
                                 p_forca_d   TYPE abap_bool
                                 p_idnrk
                                 p_matnr
                                 p_charg
                        CHANGING p_descarta  TYPE abap_bool.

  DATA ls_vermelho LIKE LINE OF it_result.

  CLEAR p_descarta.

  "1ª mensagem vermelha gerada pela verificação (amarelo 'A' e linha sem log não contam)
  LOOP AT it_result INTO DATA(ls_res) FROM p_ini_chk
                    WHERE log   IS NOT INITIAL
                      AND xloek <> 'A'.
    ls_vermelho = ls_res.
    p_descarta  = abap_true.
    EXIT.
  ENDLOOP.

  CHECK p_descarta = abap_true.

  "Dimensional passa a ser sempre vermelho (inclui divergência de tolerância)
  IF p_forca_d = abap_true.
    ls_vermelho-xloek = 'D'.
  ENDIF.

  "Remove todas as mensagens da combinação e grava só a vermelha
  DELETE it_result FROM p_ini_comb.
  APPEND ls_vermelho TO it_result.

  PERFORM f_grava_it_log_aplicacao USING p_idnrk p_matnr p_charg.

ENDFORM.                    " F_DESCARTA_COMBINACAO
